-- ============ 传送工具 ============
local function teleportTo(pos)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then print("[传送] 角色不存在") return false end
    pcall(function()
        hrp.CFrame = CFrame.new(pos + Vector3.new(0,3,0))
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    print("[传送] →", tostring(pos))
    return true
end

local tabs = {}
local currentTab = nil

local function switchTab(name)
    for tn, tab in pairs(tabs) do
        if tn == name then
            tab.Button.BackgroundColor3 = Color3.fromRGB(80,140,220)
            tab.Button.TextColor3 = Color3.new(1,1,1)
            tab.Content.Visible = true
            currentTab = name
        else
            tab.Button.BackgroundColor3 = Color3.fromRGB(45,45,65)
            tab.Button.TextColor3 = Color3.fromRGB(180,180,180)
            tab.Content.Visible = false
        end
    end
end

local function createTab(name)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0,80,1,0)
    btn.BackgroundColor3 = Color3.fromRGB(45,45,65)
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(180,180,180)
    btn.Font = Enum.Font.GothamSemibold
    btn.TextSize = 11
    btn.AutoButtonColor = false
    btn.Parent = TabBar
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0,6)
    local content = Instance.new("Frame")
    content.Size = UDim2.new(1,0,0,0)
    content.BackgroundTransparency = 1
    content.AutomaticSize = Enum.AutomaticSize.Y
    content.Visible = false
    content.Parent = Content
    local l = Instance.new("UIListLayout", content)
    l.Padding = UDim.new(0,6)
    tabs[name] = { Button = btn, Content = content }
    btn.MouseButton1Click:Connect(function() switchTab(name) end)
    if not currentTab then switchTab(name) end
    return content
end

local function createButton(parent, text, callback, color)
    local b = Instance.new("TextButton")
    b.Size = UDim2.new(1,0,0,32)
    b.BackgroundColor3 = color or Color3.fromRGB(45,45,65)
    b.Text = text
    b.TextColor3 = Color3.new(1,1,1)
    b.Font = Enum.Font.GothamSemibold
    b.TextSize = 13
    b.AutoButtonColor = false
    b.Parent = parent
    Instance.new("UICorner", b).CornerRadius = UDim.new(0,6)
    b.MouseEnter:Connect(function()
        TweenService:Create(b, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(60,60,85)}):Play()
    end)
    b.MouseLeave:Connect(function()
        TweenService:Create(b, TweenInfo.new(0.15), {BackgroundColor3 = color or Color3.fromRGB(45,45,65)}):Play()
    end)
    b.MouseButton1Click:Connect(function()
        local ok, err = pcall(callback)
        if not ok then warn("[按钮出错] " .. tostring(err)) end
    end)
    return b
end

local function createToggle(parent, text, default, callback, textSize)
    local f = Instance.new("Frame")
    f.Size = UDim2.new(1,0,0,32)
    f.BackgroundColor3 = Color3.fromRGB(45,45,65)
    f.Parent = parent
    Instance.new("UICorner", f).CornerRadius = UDim.new(0,6)
    local lb = Instance.new("TextLabel")
    lb.Size = UDim2.new(1,-60,1,0)
    lb.Position = UDim2.new(0,10,0,0)
    lb.BackgroundTransparency = 1
    lb.Text = text
    lb.TextColor3 = Color3.new(1,1,1)
    lb.Font = Enum.Font.GothamSemibold
    lb.TextSize = textSize or 13
    lb.TextXAlignment = Enum.TextXAlignment.Left
    lb.Parent = f
    local bg = Instance.new("Frame")
    bg.Size = UDim2.new(0,44,0,22)
    bg.Position = UDim2.new(1,-54,0.5,-11)
    bg.BackgroundColor3 = Color3.fromRGB(60,60,60)
    bg.Parent = f
    Instance.new("UICorner", bg).CornerRadius = UDim.new(0,4)
    local c = Instance.new("Frame")
    c.Size = UDim2.new(0,18,0,18)
    c.Position = UDim2.new(0,2,0.5,-9)
    c.BackgroundColor3 = Color3.new(1,1,1)
    c.Parent = bg
    Instance.new("UICorner", c).CornerRadius = UDim.new(0,3)
    local enabled = default or false
    local function upd()
        if enabled then
            TweenService:Create(bg,TweenInfo.new(0.2),{BackgroundColor3=Color3.fromRGB(80,200,120)}):Play()
            TweenService:Create(c,TweenInfo.new(0.2),{Position=UDim2.new(1,-20,0.5,-9)}):Play()
        else
            TweenService:Create(bg,TweenInfo.new(0.2),{BackgroundColor3=Color3.fromRGB(60,60,60)}):Play()
            TweenService:Create(c,TweenInfo.new(0.2),{Position=UDim2.new(0,2,0.5,-9)}):Play()
        end
    end
    upd()
    local ca = Instance.new("TextButton")
    ca.Size = UDim2.new(1,0,1,0)
    ca.BackgroundTransparency = 1
    ca.Text = ""
    ca.Parent = f
    ca.MouseButton1Click:Connect(function()
        enabled = not enabled
        upd()
        if callback then callback(enabled) end
    end)
    return { Set = function(_,v) enabled=v; upd() end, Get = function() return enabled end }
end

local function getBGM()
    local b = SoundService:FindFirstChild("BGM")
    if not b then
        b = Instance.new("Sound")
        b.Name = "BGM"
        b.Parent = SoundService
        b.Looped = true
        b.Playing = true
    end
    return b
end

local function extractId(sid)
    local id = string.match(sid, "rbxassetid://(%d+)")
    return id or sid
end
